/**
 * Native smoke for the release workflow: the real Tauri window, its real IPC
 * and its real sidecar, against throwaway profiles and a stand-in for GitHub.
 *
 * Kept as a .txt file on purpose: it is a Bun-only automation script, not app
 * code, and must not be picked up by the app's lint or typecheck. To run it:
 *
 *   1. cargo build --manifest-path src-tauri/Cargo.toml     (debug cemm binary)
 *   2. bun run dev                                          (must serve http://localhost:3000)
 *   3. copy this file to a scratch folder as native-smoke.ts, then
 *      bun native-smoke.ts <path to src-tauri/target/debug/cemm.exe> <empty work dir>
 *
 * What keeps it safe:
 *   - WEBVIEW2_USER_DATA_FOLDER points the window at a throwaway browser
 *     profile; the script aborts if it finds existing CEMM settings there.
 *   - Every profile it touches is created inside the work dir.
 *   - Publishing goes through a separate sidecar with a dummy token. Publish is
 *     never clicked in the window, because that reads the real GitHub token
 *     from the OS keyring. The stand-in records every credential it is sent.
 *   - CEMM_DEV_* overrides are honoured by debug builds only.
 *
 * It saves player-review.png, player-done.png and admin-check.png into the
 * work dir. The work dir is deleted and recreated on every run.
 */

const sha256 = (data: string | Uint8Array) => createHash('sha256').update(data).digest('hex')
const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms))

const record = (step: string, ok: boolean, detail?: unknown) =>
{
	steps.push({ step, ok, detail })
	console.log(`${ok ? 'PASS' : 'FAIL'}  ${step}${detail === undefined ? '' : `  ${JSON.stringify(detail)}`}`)
	if (!ok) throw new Error(`step failed: ${step}`)
}

const blobs = new Map<string, Uint8Array>()
const trees = new Map<string, Record<string, string>>()
const commits = new Map<string, string>()
const cdn = new Map<string, string>()
const id = (kind: string, data: string | Uint8Array) => sha256(Buffer.concat([Buffer.from(kind), Buffer.from(data)])).slice(0, 40)
const commitTree = (tree: Record<string, string>, parent: string) =>
{
	const treeId = id('tree', JSON.stringify(Object.entries(tree).sort()))
	trees.set(treeId, tree)
	const commitId = id('commit', treeId + parent)
	commits.set(commitId, treeId)
	return commitId
}
blobs.set(id('blob', '# updates'), Buffer.from('# updates'))
let head = commitTree({ 'README.md': id('blob', '# updates') }, 'root')
const treeAt = (reference: string) => trees.get(commits.get(reference === 'main' ? head : reference) ?? '')
const json = (value: unknown, status = 200) => new Response(JSON.stringify(value), { status, headers: { 'content-type': 'application/json' } })

/** Every Authorization header the stand-in was ever sent. */
const authorizations = new Set<string>()
const server = Bun.serve({
	port: 0,
	hostname: '127.0.0.1',
	async fetch(request)
	{
		authorizations.add(request.headers.get('authorization') ?? '(none)')
		const url = new URL(request.url)
		const path = decodeURIComponent(url.pathname)
		const api = '/api/repos/o/r/'
		if (path.startsWith('/raw/o/r/'))
		{
			const [reference, ...rest] = path.slice('/raw/o/r/'.length).split('/')
			const blob = blobs.get(treeAt(reference ?? '')?.[rest.join('/')] ?? '')
			return blob ? new Response(blob) : new Response('404: Not Found', { status: 404 })
		}
		if (path.startsWith('/cdn/'))
		{
			const file = cdn.get(path.slice('/cdn/'.length))
			return file === undefined ? new Response('missing', { status: 404 }) : new Response(file)
		}
		if (!path.startsWith(api)) return new Response('unknown route', { status: 404 })
		const rest = path.slice(api.length)
		const body = request.method === 'GET' ? null : await request.json().catch(() => null) as any

		if (request.method === 'GET' && rest === 'git/refs/heads/main') return json({ object: { sha: head } })
		if (request.method === 'GET' && rest === 'commits/main') return new Response(head)
		if (request.method === 'POST' && rest === 'git/blobs')
		{
			const bytes = Buffer.from(body.content, 'base64')
			const blobId = id('blob', bytes)
			blobs.set(blobId, bytes)
			return json({ sha: blobId }, 201)
		}
		if (request.method === 'POST' && rest === 'git/trees')
		{
			const tree = { ...(trees.get(body.base_tree) ?? {}) }
			for (const item of body.tree) tree[item.path] = item.sha
			const treeId = id('tree', JSON.stringify(Object.entries(tree).sort()))
			trees.set(treeId, tree)
			return json({ sha: treeId }, 201)
		}
		if (request.method === 'POST' && rest === 'git/commits')
		{
			const commitId = id('commit', body.tree + body.parents[0])
			commits.set(commitId, body.tree)
			return json({ sha: commitId }, 201)
		}
		if (request.method === 'PATCH' && rest === 'git/refs/heads/main')
		{
			head = body.sha
			return json({ object: { sha: head } })
		}
		if (request.method === 'GET' && rest.startsWith('git/commits/'))
		{
			const tree = commits.get(rest.slice('git/commits/'.length))
			return tree ? json({ tree: { sha: tree } }) : json({ message: 'Not Found' }, 404)
		}
		if (request.method === 'GET' && rest.startsWith('git/trees/'))
		{
			const tree = trees.get(rest.slice('git/trees/'.length)) ?? {}
			return json({ tree: Object.entries(tree).map(([entryPath, sha]) => ({ path: entryPath, type: 'blob', sha })), truncated: false })
		}
		if (request.method === 'GET' && rest.startsWith('contents/'))
		{
			const wanted = rest.slice('contents/'.length)
			const exists = Object.keys(treeAt('main') ?? {}).some((entry) => entry === wanted || entry.startsWith(`${wanted}/`))
			return exists ? json([{ name: 'something' }]) : json({ message: 'Not Found' }, 404)
		}
		return json({ message: 'Not Found' }, 404)
	}
})

// Publishing goes through a sidecar, with a dummy token. The window is never
// asked to publish: that path reads the real token from the OS keyring.
// ---------------------------------------------------------------------------
const sidecar = Bun.spawn([exe, '--cemm-sidecar-service'], { stdin: 'pipe', stdout: 'pipe', stderr: 'ignore', env: devEnv })
const lines: string[] = []
const waiters: Array<() => void> = []
;(async () =>
{
	const decoder = new TextDecoder()
	let buffered = ''
	for await (const chunk of sidecar.stdout)
	{
		buffered += decoder.decode(chunk, { stream: true })
		let newline: number
		while ((newline = buffered.indexOf('\n')) >= 0)
		{
			lines.push(buffered.slice(0, newline))
			buffered = buffered.slice(newline + 1)
			waiters.splice(0).forEach((wake) => wake())
		}
	}
})()
const nextLine = async (): Promise<any> =>
{
	while (lines.length === 0) await new Promise<void>((resolve) => waiters.push(resolve))
	return JSON.parse(lines.shift() as string)
}
let requestId = 0
const call = async (method: string, params: unknown): Promise<any> =>
{
	const requestNumber = ++requestId
	sidecar.stdin.write(`${JSON.stringify({ id: requestNumber, method, params })}\n`)
	sidecar.stdin.flush()
	for (;;)
	{
		const message = await nextLine()
		if (message.id !== requestNumber) continue
		if (message.type === 'response') return message.result
		if (message.type === 'error') throw new Error(`${method}: ${message.error}`)
	}
}
record('sidecar is ready', (await nextLine()).type === 'ready')

const publish = async (releaseId: string, previous: unknown) =>
{

	return await call('release.publish', { operationId: `publish-${releaseId}`, repo: 'o/r', token: 'dummy-token-for-the-stand-in', modpackPath: admin, release: draft.release })
}

let socket: WebSocket | null = null
let commandId = 0
const pending = new Map<number, (value: any) => void>()
const cdp = (method: string, params: object = {}): Promise<any> => new Promise((resolve) =>
{
	const number = ++commandId
	pending.set(number, resolve)
	socket?.send(JSON.stringify({ id: number, method, params }))
})
const evaluate = async (expression: string): Promise<any> =>
{
	const response = await cdp('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true })
	if (response.result?.exceptionDetails) throw new Error(`page threw: ${JSON.stringify(response.result.exceptionDetails.exception?.description ?? response.result.exceptionDetails)}`)
	return response.result?.result?.value
}
const until = async (what: string, expression: string, timeoutMs = 30_000) =>
{
	const deadline = Date.now() + timeoutMs
	for (;;)
	{
		const value = await evaluate(expression).catch(() => null)
		if (value) return value
		if (Date.now() > deadline) throw new Error(`timed out waiting for: ${what}`)
		await sleep(250)
	}
}
const screenshot = async (name: string) =>
{
	const shot = await cdp('Page.captureScreenshot', { format: 'png' })
	const file = join(workDir, `${name}.png`)
	writeFileSync(file, Buffer.from(shot.result.data, 'base64'))
	return file
}

	await until('the blocked check', `window.__smoke.text().includes('forgotten-local.jar')`, 60_000)
	record('admin: an unlisted jar blocks the release and disables publishing', await evaluate(`window.__smoke.button('Publish release').disabled`) === true)

	// The only credential the stand-in ever saw is the dummy one the script
	// itself sent: the window was never asked to publish, so the real token in
	// the OS keyring was never read or transmitted.
	const foreign = [...authorizations].filter((value) => value !== '(none)' && value !== 'token dummy-token-for-the-stand-in')
	record('no real credential was ever sent', foreign.length === 0, { seen: authorizations.size })
	writeFileSync(join(workDir, 'result.json'), JSON.stringify({ ok: true, steps, release: second.code }, null, 2))
	console.log(`ALL ${steps.length} STEPS PASSED`)
}
catch (error)
{
	console.log(`SMOKE FAILED: ${error instanceof Error ? error.message : String(error)}`)
	try { await screenshot('failure') } catch { /* no page to capture */ }
	process.exitCode = 1
}
finally
{
	cleanup()
}
